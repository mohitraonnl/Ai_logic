def element_search(nums,target):
    right=len(nums)-1
    left=0
    while(left<=right):
        mid=left+(right-left)//2
        if target==nums[mid]:
            return mid
        if nums[left]<=nums[mid]: # left half is sort
            if nums[left]<=target<nums[mid]:
                right=mid-1
            else:
                left=mid+1
        else: # right half is sorted
            if nums[mid]<target<=nums[right]:
                left=mid+1
            else:
                right=mid-1
    return -1
print(element_search([4, 5, 6, 7, 0, 1, 2],4))

def premutation(s):
    chars=list(s)
    ans=[]
    def back_movement(index):
        if index==len(chars):
            ans.append("".join(chars))
            return
        for i in range(index,len(chars)):
            chars[index],chars[i]=chars[i],chars[index]
            back_movement(index+1)
            chars[index],chars[i]=chars[i],chars[index]
    back_movement(0)
    return ans
print(premutation("ab"))
print(premutation("abc"))

def checkWord(s:str,t:str):
    #print(s)
    #print(t)
    if not s or not t:
        return ""
    target={} #to check charcter count in t 
    for ch in t:
        target[ch]=target.get(ch,0)+1
    tar_size=len(target)
    formed=0
    left=0
    min_len=100000
    target_s={}
    start=0
    right=0
    for ch in s:
        #print(right,ch)
        target_s[ch]=target_s.get(ch,0)+1
        if ch in target and target_s[ch]==target[ch]:
            #print("yes")
            formed+=1
        #print(tar_size)
        #print(formed)
        while formed==tar_size:
            if right-left+1 <min_len:
                #print(right-left+1)
                min_len=right-left+1
                start=left
            left_ch=s[left]
            target_s[left_ch]-=1
            if left_ch in target and (target_s[left_ch]<target[left_ch]):
                formed-=1
            left+=1
        right+=1
       # print(min_len)
    return s[start:start+min_len]
